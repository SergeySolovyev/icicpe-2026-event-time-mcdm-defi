\begin{frame}{Motivation \& research question}
  \textbf{Setting.} A USDC supplier can sit in Aave V3 or Compound V3. Rates
  move with utilization and governance; the protocols are cointegrated but
  lead/lag each other.

  \medskip
  \textbf{Standard practice} reacts to the \emph{current} (EMA-smoothed)
  rate spread --- it always lags a regime change.

  \medskip
  \textbf{Research question.} Does replacing the reactive rate signal with a
  \textbf{learned 12-hour-ahead forecast} produce a better risk-adjusted
  allocation?

  \medskip
  \begin{block}{Pre-registered hypotheses}
    \begin{itemize}
      \item \textbf{H1:} predictive MCDM beats reactive MCDM-EMA by
            $\Delta\text{Sharpe}\ge 0.2$ on the held-out window.
      \item \textbf{H0:} no improvement --- a \emph{publishable negative
            result} (methodological-transfer falsification).
    \end{itemize}
  \end{block}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{The empirical anchor: a real regime shift}
  Joined hourly panel, \textbf{12{,}895 overlapping hours}, $\sim$98.5\%
  coverage of Nov 2024 -- Apr 2026 (assembled \emph{before} test-window
  exposure).

  \medskip
  \begin{center}
  \begin{tabular}{lccc}
    \toprule
    Quarter & median spread & vol (std) & Aave-pays-more \\
    \midrule
    2025 Q3 & $-0.22$ pp & $1.40$ pp & 39.9\% \\
    \textbf{2025 Q4} & $\mathbf{+0.18}$ \textbf{pp} & $2.02$ pp & \textbf{56.3\%} \\
    \bottomrule
  \end{tabular}
  \end{center}

  \medskip
  \begin{itemize}
    \item \textbf{First quarter Aave systematically pays more} --- a 16-point
          regime flip; the median spread crosses zero.
    \item Volatility regimes vary $\sim$10$\times$ (5.75 pp in Q4'24
          $\to$ 0.57 pp in Q1'26).
    \item Only 40.7\% Aave-higher overall $\Rightarrow$ alpha is in
          \emph{timing crossovers}, not a structural bias --- exactly what a
          forecast should exploit.
  \end{itemize}
\end{frame}

\begin{frame}{Methodology I --- the forecaster}
  \textbf{DA-BiGRU-CNN}, dual-branch, $\sim$319K params (ported from the
  author's LOB mid-price model):

  \medskip
  \begin{itemize}
    \item \textbf{Branch A --- residual.} Subtract the protocol-known kink
          curve $f_{\text{kink}}(u)$; the net learns only
          $\varepsilon = r - f_{\text{kink}}(u)$, freeing capacity for the
          stochastic part.
    \item \textbf{Branch B --- context.} Utilization, TVL momentum, gas,
          time-of-day.
    \item \textbf{Late fusion} via multi-scale Conv1d (kernels 3/5/7),
          12-hour-ahead head.
    \item \textbf{Composite loss:}
          $\alpha\,\text{MSE} + \beta\,(1-\text{WPearson}) +
           \gamma\,\text{QuantileLoss}_{0.9}$.
  \end{itemize}

  \medskip
  \footnotesize\textcolor{gray}{Trained on Colab H100; exported to ONNX with
  torch$\leftrightarrow$onnxruntime parity verified ($\max|\Delta|\!<\!10^{-7}$).}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Methodology II --- the allocator}
  \textbf{4-factor MCDM} (SAW), ported from the author's ERC-4626 yield-vault
  paper:
  \[
    \text{Score}_i = 0.40\,f_{\text{APY}} + 0.25\,f_{\text{Risk}}
                    + 0.20\,f_{\text{Cost}} + 0.15\,f_{\text{Stab}}
  \]
  \begin{itemize}
    \item The \textbf{only} change vs.\ the reactive baseline: $f_{\text{APY}}$
          is fed the \textbf{forecast} $\hat r_{t+12h}$ instead of the
          EMA-smoothed spot rate. Everything else is held fixed --- a clean
          ablation of the forecast signal itself.
    \item \textbf{Hysteresis + cooldown} rebalance trigger (gas-aware) to
          suppress churn.
    \item Implemented as \texttt{PredictiveMCDMStrategy} on a new
          \texttt{BaseLendingAllocationStrategy} abstraction.
  \end{itemize}
\end{frame}

\begin{frame}{Experimental design}
  \begin{columns}[T]

      \textbf{Data}
      \begin{itemize}
        \item Aave V3 + Compound V3 USDC, hourly, 18 months
        \item Strict \emph{block-number} train/val/test split
        \item Test window: 2026 Q1--Q2 (held out)
      \end{itemize}
      \medskip
      \textbf{Significance}
      \begin{itemize}
        \item 1000-iter paired \emph{monthly-Sharpe} bootstrap
        \item McNemar on directional calls
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Comparators}
      \begin{itemize}
        \item 4 baselines: buy-hold, APY-greedy, MCDM-EMA, MCDM-CIR
        \item \textbf{15 ablations} (no-forecast, single-branch,
              no-kink, loss terms, horizon sweep, OOD depeg, \dots)
        \item Regime-conditional reporting (per-quarter)
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Contributions \& conclusion}
  \begin{itemize}
    \item \textbf{Method.} A clean, single-signal test of forecast- vs.\
          reaction-driven cross-protocol allocation, with pre-registered
          H0/H1 and regime-conditional evaluation.
    \item \textbf{Empirical.} Documented + corrected a real 2025 Q3$\to$Q4
          Aave/Compound regime flip on 98.5\%-coverage mainnet data
          (and a methodological lesson: sparse sampling fabricated a false
          earlier ``shift'').
    \item \textbf{Upstream (\texttt{fractal-defi}).}
      \begin{itemize}
        \item \textbf{Extra+1:} Compound V3 loader + uniform
              \texttt{utilization} field on both lending states.
        \item \textbf{Extra+2:} \texttt{BaseLendingAllocationStrategy}
              abstraction; \texttt{PredictiveMCDMStrategy} as first subclass.
      \end{itemize}
    \item \textbf{Reproducible.} One-command pipeline; numbers on this deck
          and in the paper flow from the same backtest CSVs.
